Thuật toán sử dụng:
Mfa2 = Fa2 · (dw2 / 2)
BD = BC + CD
FDy = (Fr2·BC - Fd·AB - Mfa2) / BD
FBy = Fd + Fr2 - FDy

Qy_AB = -Fd
Qy_BC = Qy_AB + FBy
Qy_CD = Qy_BC - Fr2

Mx(A)=0
Mx(B)=Qy_AB·AB
Mx(C-trái)=Qy_AB·AB + Qy_BC·BC
Mx(C-phải)=Mx(C-trái) - Mfa2
Mx(D)=Mx(C-phải) + Qy_CD·CD

FDx = (Ft2·BC) / BD
FBx = Ft2 - FDx

Qx_AB = 0
Qx_BC = -FBx
Qx_CD = Qx_BC + Ft2

My(A)=0
My(B)=Qx_AB·AB
My(C)=Qx_AB·AB + Qx_BC·BC
My(D)=Qx_AB·AB + Qx_BC·BC + Qx_CD·CD
